\chapter{Diagnostics}
\label{app:diagnostic-reports}

\begin{verbatim}
cargo run -- parse "(a|ab)(b|)" <INPUT> --parser <PARSER> --diag 3
\end{verbatim}

\begin{center}
\begin{tabular}{@{}l>{\raggedright\arraybackslash}p{10.5cm}@{}}
\hline
Field & Value \\
\hline
Expression & \texttt{"(a|ab)(b|)"}, that is $(a+ab)(b+\varepsilon)$ \\
Inputs & \texttt{"ab"} (match), \texttt{"b"} (no match) \\
Parsers & \texttt{deriv\_std\_rec}, \texttt{deriv\_bc}, \texttt{pderiv\_std}, \texttt{pderiv\_bc} \\
Output file & \texttt{reports/report.txt} \\
\hline
Omitted & the \texttt{Timestamp} line \\
Wrapped & long lines \\
$\partial$ & \texttt{d} \\
$\cdot$ & \texttt{.} \\
$\oplus$ & \texttt{(+)} \\
$\varepsilon$ & \texttt{epsilon} \\
$\phi$ & \texttt{phi} \\
$\checkmark$ & \texttt{[OK]} \\
\texttt{Policy} & the implementation's internal label, not a claim about
                  disambiguation policy \\
\hline
\end{tabular}
\end{center}
